sequenceDiagram
    autonumber
    actor U as Boutique (utilisateur)
    participant V as I:SubscriptionPlansView
    participant Auth as C:AuthContext
    participant API as C:ApiClient
    participant C as C:SubscriptionController
    participant EP as E:SubscriptionPlan
    participant EB as E:BoutiqueSubscription

    rect rgb(230, 244, 255)
    Note over U,EB: Chargement des plans
    activate V
    U->>V: Naviguer vers /subscription
    V->>Auth: useAuth() (user, subscriptionAccess, refreshSubscriptionAccess)
    Auth-->>V: contexte
    V->>V: useEffect → loadData()
    V->>V: setLoading(true)
    V-->>U: "Loading plans..."

    V->>API: getSubscriptionPlans()
    activate API
    API->>C: GET /api/boutiques/subscription-plans
    activate C
    C->>EP: find({isActive:true})
    activate EP
    EP-->>C: Plans
    deactivate EP
    alt Erreur
        C-->>API: HTTP 4xx/5xx
        API-->>V: throw
        V-->>U: Toast "Failed to load subscription plans"
    else Succès
        C-->>API: HTTP 200 {data:{plans:[...]}}
        API-->>V: response
        V->>V: setPlans(fetchedPlans)
        V->>V: sortedPlans = plans.sort(displayOrder)
        V->>V: setSelectedPlanId(sortedPlans[1]?._id || sortedPlans[0]._id)

        V->>Auth: refreshSubscriptionAccess()
        Auth->>API: getBoutiqueSubscriptionAccess(boutiqueId)
        API->>C: GET /api/boutiques/{boutiqueId}/subscription/access
        C->>EB: findOne({boutiqueId})
        activate EB
        EB-->>C: subscription (ou null)
        deactivate EB
        C-->>API: {hasManagementAccess, subscription}
        API-->>Auth: response
        Auth-->>V: subscriptionAccess mis à jour
        V-->>U: Afficher 3 cartes (Get started / Recommandé / Premium) + 2 stat cards
    end
    deactivate C
    deactivate API
    V->>V: setLoading(false)
    deactivate V
    end
