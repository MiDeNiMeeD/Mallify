sequenceDiagram
    autonumber
    actor U as Utilisateur
    participant V4 as I:CreateNewPasswordView
    participant V5 as I:SignInView
    participant C as C:AuthController
    participant Cache as E:PasswordResetCache
    participant EU as E:User

    rect rgb(255, 235, 235)
    Note over U,EU: Étape 3 — Création du nouveau mot de passe
    activate V4
    V4-->>U: Interface affichée
    U->>V4: Saisir nouveau mot de passe
    U->>V4: Saisir confirmation
    U->>V4: Cliquer "Continue"
    V4->>V4: valider(champs non vides, newPassword === confirm, email & code présents)
    alt Champs manquants
        V4-->>U: Alerte "Please enter password and confirm it"
    else Mots de passe différents
        V4-->>U: Alerte "Passwords do not match"
    else email ou code manquant
        V4-->>U: Alerte "Missing email or verification code"
    else Validation OK
        V4->>C: POST /auth/reset-password {email, code, newPassword}
        activate C
        C->>Cache: récupérer(email)
        activate Cache
        Cache-->>C: codeAttendu + expiration
        deactivate Cache
        alt Code OTP invalide ou expiré
            C-->>V4: HTTP 400 {message}
            V4-->>U: Alerte "Could not reset password"
        else Code OTP valide
            C->>C: hacher(newPassword)
            C->>EU: mettreAJourMotDePasse(email, nouveauHash)
            activate EU
            EU-->>C: OK
            deactivate EU
            C->>Cache: supprimer(email)
            C-->>V4: HTTP 200
            V4-->>U: Alerte "Password reset successfully"
            V4->>V5: navigation.navigate("Login")
            deactivate V4
        end
        deactivate C
    end
    end
